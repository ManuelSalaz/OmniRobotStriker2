45// Configuración (CONF). Para evitar problemas de bitfields/endianness en distintas toolchains,
46// usa el campo WORD directamente. Los enums te ayudan a construir el WORD.
47typedef union {
48 uint16_t WORD;
49 struct __attribute__((packed)) {
50 uint16_t PM : 2; // power mode
51 uint16_t HYST : 2; // hysteresis
52 uint16_t OUTS : 2; // output stage (solo pin OUT)
53 uint16_t PWMF : 2; // PWM freq (solo pin OUT)
54 uint16_t SF : 2; // slow filter
55 uint16_t FTH : 3; // fast filter threshold
56 uint16_t WD : 1; // watchdog
57 uint16_t RESERVED : 2;
58 };
59} AS5600_config_t;
